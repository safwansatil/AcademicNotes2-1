// SimpleXmlWriter.java - Writes object to XML file
import java.io.*;

public class SimpleXmlWriter {
    
    public static void savePersonToXml(Person person, String filename) {
        try (PrintWriter writer = new PrintWriter(filename)) {
            writer.println("<?xml version=\"1.0\"?>");
            writer.println("<Person>");
            writer.println("  <name>" + escapeXml(person.getName()) + "</name>");
            writer.println("  <age>" + person.getAge() + "</age>");
            writer.println("  <email>" + escapeXml(person.getEmail()) + "</email>");
            writer.println("</Person>");
            System.out.println("Saved to " + filename);
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
    
    private static String escapeXml(String input) {
        // Replace special XML characters
        return input.replace("&", "&amp;")
                   .replace("<", "&lt;")
                   .replace(">", "&gt;")
                   .replace("\"", "&quot;")
                   .replace("'", "&apos;");
    }
}

// SimpleXmlReader.java - Reads XML file to object
import java.io.*;
import java.util.regex.*;

public class SimpleXmlReader {
    
    public static Person loadPersonFromXml(String filename) {
        Person person = new Person();
        
        try (BufferedReader reader = new BufferedReader(new FileReader(filename))) {
            String line;
            while ((line = reader.readLine()) != null) {
                line = line.trim();
                
                // Extract name
                if (line.startsWith("<name>")) {
                    String name = extractValue(line);
                    person.setName(unescapeXml(name));
                }
                // Extract age
                else if (line.startsWith("<age>")) {
                    String ageStr = extractValue(line);
                    person.setAge(Integer.parseInt(ageStr));
                }
                // Extract email
                else if (line.startsWith("<email>")) {
                    String email = extractValue(line);
                    person.setEmail(unescapeXml(email));
                }
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
        
        return person;
    }
    
    private static String extractValue(String line) {
        // Extract text between tags: <tag>value</tag>
        Pattern pattern = Pattern.compile(">(.*?)<");
        Matcher matcher = pattern.matcher(line);
        if (matcher.find()) {
            return matcher.group(1);
        }
        return "";
    }
    
    private static String unescapeXml(String input) {
        // Restore special characters
        return input.replace("&amp;", "&")
                   .replace("&lt;", "<")
                   .replace("&gt;", ">")
                   .replace("&quot;", "\"")
                   .replace("&apos;", "'");
    }
}
